💻 Production Implementation Explorer

import { create } from "zustand";
import { persist, createJSONStorage } from "zustand/middleware";

export const useCheckoutStore = create()(
  persist(
    (set, get) => ({
      step: 1,
      merchantSlug: "",
      sessionData: null,
      confirmedPaymentAmount: 0,

      setStep: (step) => set({ step }),
      setSessionData: (sessionData) => set({ sessionData }),
      resetForNewPayment: () => set({ step: 2, sessionData: null }),
      submitPaymentData: () => set({ step: 3 }),
    }),
    {
      name: "bitnormous_checkout_state",
      storage: createJSONStorage(() => localStorage),
    }
  )
);
